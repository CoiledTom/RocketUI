-- 2. CARREGAMENTO DE ÍCONES DINÂMICO COM FALLBACK
-- ==================================================
local IconsData = {}
local ICONS_URL = "https://raw.githubusercontent.com/CoiledTom/RocketUI/refs/heads/main/Icons.lua"

task.spawn(function()
    local success, response = pcall(function()
        return game:HttpGet(ICONS_URL)
    end)
    if success and response then
        local loadSuccess, loadedIcons = pcall(function()
            return loadstring(response)()
        end)
        if loadSuccess and type(loadedIcons) == "table" then
            IconsData = loadedIcons
        end
    end
end)

function RocketUI:GetIcon(name)
    if type(name) == "string" and IconsData[name] then
        return IconsData[name]
    end
    return nil
end

function RocketUI:CreateIcon(parent, name, size, color)
    size = size or 18
    color = color or RocketUI.Themes[RocketUI.CurrentTheme].Text

    local iconData = RocketUI:GetIcon(name)
    
    if iconData then
        local image = Instance.new("ImageLabel")
        image.Name = "Icon_" .. tostring(name)
        image.BackgroundTransparency = 1
        image.Size = UDim2.fromOffset(size, size)
        image.AnchorPoint = Vector2.new(0.5, 0.5)
        image.Position = UDim2.fromScale(0.5, 0.5)
        image.ImageColor3 = color
        image.Parent = parent

        if type(iconData) == "string" then
            image.Image = iconData
        elseif type(iconData) == "table" then
            image.Image = iconData.Image or iconData.id or ""
            if iconData.RectMin and iconData.RectSize then
                image.ImageRectOffset = iconData.RectMin
                image.ImageRectSize = iconData.RectSize
            end
        end
        return image
    else
        local fallback = Instance.new("TextLabel")
        fallback.Name = "Icon_Fallback"
        fallback.BackgroundTransparency = 1
        fallback.Size = UDim2.fromOffset(size, size)
        fallback.Font = Enum.Font.GothamBold
        fallback.Text = string.sub(tostring(name or "?"), 1, 1):upper()
        fallback.TextColor3 = color
        fallback.TextSize = math.floor(size * 0.7)
        fallback.Parent = parent
        return fallback
    end
end

-- 3. HELPER DE DRAG E SUPORTE A TOUCH / MOUSE
-- ==================================================
local function TrackConnection(connection)
    table.insert(RocketUI.Connections, connection)
    return connection
end

local function Tween(object, info, properties)
    local tween = TweenService:Create(object, info, properties)
    tween:Play()
    return tween
end

local function EnableDragging(dragHandle, frameToMove)
    local dragging = false
    local dragInput, dragStart, startPos

    local function update(input)
        local delta = input.Position - dragStart
        frameToMove.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end

    TrackConnection(dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frameToMove.Position

            local endedConn
            endedConn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    endedConn:Disconnect()
                end
            end)
        end
    end))

    TrackConnection(dragHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))

    TrackConnection(UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            update(input)
        end
    end))
end

    contentLabel.Size = UDim2.new(1, -45, 0, 0)
    contentLabel.AutomaticSize = Enum.AutomaticSize.Y
    contentLabel.Position = UDim2.new(0, 40, 0, 26)
    contentLabel.BackgroundTransparency = 1
    contentLabel.Font = Enum.Font.Gotham
    contentLabel.Text = content
    contentLabel.TextColor3 = theme.Muted

    contentLabel.TextXAlignment = Enum.TextXAlignment.Left
    contentLabel.Parent = card

    padding.PaddingBottom = UDim.new(0, 12)
    padding.Parent = card

    progressBar.Size = UDim2.new(1, 0, 0, 3)
    progressBar.Position = UDim2.new(0, 0, 1, -3)

        tw.Completed:Connect(function()
            card:Destroy()
        end)
    end

    closeBtn.MouseButton1Click:Connect(dismiss)
    task.delay(duration, dismiss)

        -- COMPONENTE: SLIDER
        function TabObject:Slider(sOpts)
            sOpts = sOpts or {}
            local sTitle = sOpts.Title or "Slider"
            local min = sOpts.Min or 0
            local max = sOpts.Max or 100
            local default = math.clamp(sOpts.Default or min, min, max)
            local increment = sOpts.Increment or 1
            local callback = sOpts.Callback or function() end

            local card = Instance.new("Frame")
            card.Size = UDim2.new(1, 0, 0, 46)
            card.BackgroundColor3 = theme.Card
            card.Parent = TabContainer

            local cCorner = Instance.new("UICorner")
            cCorner.CornerRadius = UDim.new(0, 6)
            cCorner.Parent = card

            local label = Instance.new("TextLabel")
            label.Size = UDim2.new(1, -60, 0, 20)
            label.Position = UDim2.new(0, 12, 0, 4)
            label.BackgroundTransparency = 1
            label.Font = Enum.Font.GothamMedium
            label.Text = sTitle
            label.TextColor3 = theme.Text
            label.TextSize = 12
            label.TextXAlignment = Enum.TextXAlignment.Left
            label.Parent = card

            local valLabel = Instance.new("TextLabel")
            valLabel.Size = UDim2.new(0, 50, 0, 20)
            valLabel.Position = UDim2.new(1, -62, 0, 4)
            valLabel.BackgroundTransparency = 1
            valLabel.Font = Enum.Font.GothamBold
            valLabel.Text = tostring(default)
            valLabel.TextColor3 = theme.Accent
            valLabel.TextSize = 12
            valLabel.TextXAlignment = Enum.TextXAlignment.Right
            valLabel.Parent = card

            local sliderTrack = Instance.new("Frame")
            sliderTrack.Size = UDim2.new(1, -24, 0, 6)
            sliderTrack.Position = UDim2.new(0, 12, 0, 28)
            sliderTrack.BackgroundColor3 = theme.Hover
            sliderTrack.Parent = card

            local trackCorner = Instance.new("UICorner")
            trackCorner.CornerRadius = UDim.new(1, 0)
            trackCorner.Parent = sliderTrack

            local alpha = (default - min) / (max - min)

            local sliderFill = Instance.new("Frame")
            sliderFill.Size = UDim2.new(alpha, 0, 1, 0)
            sliderFill.BackgroundColor3 = theme.Accent
            sliderFill.Parent = sliderTrack

            local fillCorner = Instance.new("UICorner")
            fillCorner.CornerRadius = UDim.new(1, 0)
            fillCorner.Parent = sliderFill

            local isDragging = false

            local function UpdateSlider(input)
                local trackAbsPos = sliderTrack.AbsolutePosition.X
                local trackAbsSize = sliderTrack.AbsoluteSize.X
                local mousePos = input.Position.X

                local rel = math.clamp((mousePos - trackAbsPos) / trackAbsSize, 0, 1)
                local rawValue = min + (max - min) * rel
                local steppedValue = math.floor((rawValue / increment) + 0.5) * increment
                steppedValue = math.clamp(steppedValue, min, max)

                local newAlpha = (steppedValue - min) / (max - min)
                sliderFill.Size = UDim2.new(newAlpha, 0, 1, 0)
                valLabel.Text = tostring(steppedValue)

                pcall(callback, steppedValue)
            end

            TrackConnection(sliderTrack.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    isDragging = true
                    UpdateSlider(input)
                end
            end))

            TrackConnection(UserInputService.InputChanged:Connect(function(input)
                if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    UpdateSlider(input)
                end
            end))

            TrackConnection(UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    isDragging = false
                end
            end))
        end
